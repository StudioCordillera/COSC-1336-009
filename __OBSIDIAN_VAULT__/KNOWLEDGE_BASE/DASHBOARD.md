def get_valid_input(prompt, data_type=str, min_val=None, max_val=None):
    """Professor's standard validation pattern"""
    while True:
        try:
            value = data_type(input(prompt).strip())
            if min_val and value < min_val:
                print(f"Error: Must be at least {min_val}")
                continue
            if max_val and value > max_val:
                print(f"Error: Cannot exceed {max_val}")
                continue
            return value
        except ValueError:
            print("Error: Invalid input type")
        except KeyboardInterrupt:
            return None
